        /* Tạo HTML chuẩn Flatsome cho từng sản phẩm:
           - Có % giảm giá góc trên bên trái (.flatsome-discount-badge)
           - Không gắn thêm các nhãn dán rườm rà khác
           - Tên sản phẩm viết đầy đủ, tự xuống dòng gọn gàng, không bị cắt chữ hay tràn ra ngoài */
        function createProductCardHTML(item) {
            const imgPrimary = `https://lh3.googleusercontent.com/d/${item.id}=w600`;
            const imgFallback = `https://drive.google.com/thumbnail?id=${item.id}&sz=w600`;

            return `
                <div class="product-card bg-white rounded-lg border border-slate-200 overflow-hidden flex flex-col justify-between" data-name="${item.name.toLowerCase()}">
                    <div>
                        <!-- Khung ảnh sản phẩm + % giảm giá góc trên bên trái chuẩn Flatsome -->
                        <div class="relative w-full aspect-square bg-slate-100 overflow-hidden cursor-pointer" onclick="openProductModal('${item.id}')">
                            <span class="flatsome-discount-badge">${item.discount}</span>
                            <img src="${imgPrimary}" 
                                 onerror="this.onerror=null; this.src='${imgFallback}';" 
                                 alt="${item.name}" 
                                 loading="lazy"
                                 class="w-full h-full object-cover hover:scale-105 transition duration-300">
                        </div>

                        <!-- Thông tin tên sản phẩm (Không dùng truncate để không bao giờ bị thiếu chữ) -->
                        <div class="p-2.5 sm:p-3">
                            <h3 onclick="openProductModal('${item.id}')" class="text-xs sm:text-[13px] font-bold text-slate-800 hover:text-brand cursor-pointer leading-snug break-words whitespace-normal">
                                ${item.name}
                            </h3>
                        </div>
                    </div>

                    <!-- Giá gốc + Giá giảm kiểu Flatsome -->
                    <div class="px-2.5 pb-2.5 sm:px-3 sm:pb-3 pt-1 border-t border-slate-100 flex flex-col gap-1.5">
                        <div class="flex flex-wrap items-baseline gap-x-1.5 gap-y-0.5">
                            <del class="text-[11px] text-slate-400 font-normal">${item.oldPrice}</del>
                            <ins class="text-xs sm:text-sm font-extrabold text-brand-accent no-underline">${item.newPrice}</ins>
                        </div>
                        <a href="#lien-he-bao-gia" onclick="prefillProductQuote('${item.name}')" class="w-full py-1.5 rounded bg-slate-100 hover:bg-brand hover:text-white text-brand text-[11px] font-bold text-center transition">
                            Nhận báo giá
                        </a>
                    </div>
                </div>
            `;
        }

        function renderAllProducts() {
            productsData.forEach(item => {
                const container = document.getElementById(item.section);
                if (container) {
                    container.insertAdjacentHTML('beforeend', createProductCardHTML(item));
                }
            });
        }

        /* Tìm kiếm nhanh sản phẩm trên trang */
        function filterProductsBySearch(keyword) {
            const q = (keyword || '').trim().toLowerCase();
            const cards = document.querySelectorAll('.product-card');
            cards.forEach(card => {
                const name = card.getAttribute('data-name') || '';
                if (!q || name.includes(q)) {
                    card.style.display = 'flex';
                } else {
                    card.style.display = 'none';
                }
            });
        }

        /* Modal xem ảnh thực tế lớn */
        function openProductModal(id) {
            const item = productsData.find(p => p.id === id);
            if (!item) return;
            const modal = document.getElementById('productModal');
            const img = document.getElementById('modalImg');
            document.getElementById('modalTitle').textContent = item.name;
            document.getElementById('modalPriceOld').textContent = item.oldPrice;
            document.getElementById('modalPriceNew').textContent = item.newPrice;
            document.getElementById('modalDiscount').textContent = 'Giảm ' + item.discount;
            img.src = `https://lh3.googleusercontent.com/d/${item.id}=w1000`;
            img.onerror = function() {
                this.onerror = null;
                this.src = `https://drive.google.com/thumbnail?id=${item.id}&sz=w1000`;
            };
            modal.classList.remove('hidden');
            modal.classList.add('flex');
        }

        function closeProductModal() {
            const modal = document.getElementById('productModal');
            modal.classList.add('hidden');
            modal.classList.remove('flex');
        }

        /* Slider 2 Banner tự trượt chuyên nghiệp */
        let currentBannerIndex = 0;
        let bannerTimer = null;

        function showBannerSlide(index) {
            const slide0 = document.getElementById('bannerSlide0');
            const slide1 = document.getElementById('bannerSlide1');
            const dot0 = document.getElementById('bannerDot0');
            const dot1 = document.getElementById('bannerDot1');

            currentBannerIndex = index;
            if (index === 0) {
                slide0.classList.remove('opacity-0', 'pointer-events-none', 'z-0');
                slide0.classList.add('opacity-100', 'z-10');
                slide1.classList.remove('opacity-100', 'z-10');
                slide1.classList.add('opacity-0', 'pointer-events-none', 'z-0');

                dot0.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                dot1.className = 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
            } else {
                slide1.classList.remove('opacity-0', 'pointer-events-none', 'z-0');
                slide1.classList.add('opacity-100', 'z-10');
                slide0.classList.remove('opacity-100', 'z-10');
                slide0.classList.add('opacity-0', 'pointer-events-none', 'z-0');

                dot1.className = 'w-6 h-2 rounded-full bg-amber-400 transition-all';
                dot0.className = 'w-2.5 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
            }
        }

        function nextBannerSlide() {
            showBannerSlide(currentBannerIndex === 0 ? 1 : 0);
            resetBannerTimer();
        }

        function prevBannerSlide() {
            showBannerSlide(currentBannerIndex === 0 ? 1 : 0);
            resetBannerTimer();
        }

        function goToBannerSlide(idx) {
            showBannerSlide(idx);
            resetBannerTimer();
        }

        function resetBannerTimer() {
            if (bannerTimer) clearInterval(bannerTimer);
            bannerTimer = setInterval(() => {
                showBannerSlide(currentBannerIndex === 0 ? 1 : 0);
            }, 4500);
        }

        /* Lọc danh mục bài viết trang chủ */
        function filterArticles(cat) {
            const items = document.querySelectorAll('.article-item');
            const tabs = document.querySelectorAll('.article-tab');
            tabs.forEach(t => {
                if (t.getAttribute('data-cat') === cat) {
                    t.className = 'article-tab cat-nowrap px-2.5 py-1 rounded bg-brand text-white font-semibold';
                } else {
                    t.className = 'article-tab cat-nowrap px-2.5 py-1 rounded bg-slate-100 text-slate-700 hover:bg-slate-200';
                }
            });
            items.forEach(item => {
                if (cat === 'all' || item.getAttribute('data-cat') === cat) {
                    item.style.display = 'flex';
                } else {
                    item.style.display = 'none';
                }
            });
        }
